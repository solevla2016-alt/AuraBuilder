# Зависимости Control Plane (Django).
#
# Лицензии проверяются офлайн: tools/license_gate_py.py
# и docs/TZ-GAPS.md (раздел «Исключения из п.7.1.6»).
#
# Драйвер PostgreSQL — psycopg (LGPL-3.0-only), ИСКЛЮЧЕНИЕ из п.7.1.6,
# утверждённое заказчиком 02.10.2026. Причина: Django не имеет драйвера
# на чистом Python — pg8000 лицензионно чист, но бэкенд Django его не
# поддерживает (требует psycopg или psycopg2). Альтернатива — писать
# собственный бэкенд поверх pg8000, что дороже и рискованнее.
# Основание по существу: LGPL слабее GPL, при динамической линковке
# закрытый код раскрывать не требуется.

Django==6.1.1
djangorestframework==3.18.1
psycopg[binary]==3.3.6

# CORS для разработки (редактор на 5173, Control Plane на 8000).
# В продакшене фронт и API отдаются с одного домена — пакет не нужен.
django-cors-headers==4.9.0

# WSGI-сервер и раздача статики. Оба нужны только в контейнере:
# whitenoise отдаёт /static прямо из приложения, иначе статику должен
# отдавать отдельный сервис, а лишний сервис в контейнерах — лишняя
# точка отказа. Лицензии MIT, проверяются tools/license_gate_py.py.
gunicorn==23.0.0
whitenoise==6.11.0

# Аутентификация (ТЗ п.7.7): JWT с короткоживущим access-токеном и
# refresh с ротацией. simplejwt реализует это без самописной криптографии.
djangorestframework-simplejwt==5.5.1

# Argon2id для хэширования паролей — требование ТЗ п.7.3, а не склонность:
# PBKDF2 быстрее подбирается на GPU. Без этого пакета Django не знает
# про Argon2Hasher, даже если он есть в stdlib-совместимом виде.
argon2-cffi==25.1.0
